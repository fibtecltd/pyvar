SCRIPT
import asyncio
from sqlalchemy import text
from storage.session import get_sessionmaker

QUERIES = {
    "api_usage coverage": """
        SELECT min(created_at), max(created_at), count(*) FROM api_usage""",
    "per function, status 200, last 30 days, at least 5 rows (ms)": """
        SELECT domain, function_name, count(*) AS n,
               round(percentile_cont(0.5) WITHIN GROUP (ORDER BY duration_ms)::numeric, 0) AS p50,
               round(percentile_cont(0.95) WITHIN GROUP (ORDER BY duration_ms)::numeric, 0) AS p95,
               max(duration_ms) AS max_ms
        FROM api_usage WHERE status = 200 AND created_at >= now() - interval '30 days'
        GROUP BY 1, 2 HAVING count(*) >= 5 ORDER BY p95 DESC""",
    "functions seen in 30 days with fewer than 5 successful rows": """
        SELECT count(*) FROM (SELECT 1 FROM api_usage
          WHERE status = 200 AND created_at >= now() - interval '30 days'
          GROUP BY domain, function_name HAVING count(*) < 5) t""",
}


async def main():
    async with get_sessionmaker()() as session:
        await session.execute(text("SET TRANSACTION READ ONLY"))
        await session.execute(text("SET LOCAL statement_timeout = '10s'"))
        for name, query in QUERIES.items():
            rows = (await session.execute(text(query))).all()
            print("##", name, flush=True)
            for row in rows:
                print("  ", tuple(str(v) for v in row), flush=True)
            if not rows:
                print("   (no rows)", flush=True)
        await session.rollback()


asyncio.run(main())
